// lib/admin_page.dart
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:animate_do/animate_do.dart';
import 'profile_manager.dart';

class AdminPage extends StatefulWidget {
  const AdminPage({Key? key}) : super(key: key);
  @override
  State<AdminPage> createState() => _AdminPageState();
}

class _AdminPageState extends State<AdminPage>
    with SingleTickerProviderStateMixin {

  // ── palette ──────────────────────────────────────────────
  static const Color _navy    = Color(0xFF0D2137);
  static const Color _blue    = Color(0xFF1565C0);
  static const Color _surface = Color(0xFFF4F7FB);

  late TabController _tabController;
  final List<String> _tabs = ['Overview', 'Users', 'Orders'];

  // user search
  final _searchController = TextEditingController();
  String _searchQuery = '';
  String _roleFilter  = 'All';

  @override
  void initState() {
    super.initState();
    _tabController = TabController(length: _tabs.length, vsync: this);
    _searchController.addListener(
            () => setState(() => _searchQuery = _searchController.text.toLowerCase()));
  }

  @override
  void dispose() {
    _tabController.dispose();
    _searchController.dispose();
    super.dispose();
  }

  Future<void> _logout() async {
    await FirebaseAuth.instance.signOut();
    if (mounted) Navigator.pushReplacementNamed(context, '/login');
  }

  void _snack(String msg, {bool error = false}) {
    ScaffoldMessenger.of(context).showSnackBar(SnackBar(
      content: Row(children: [
        Icon(
          error ? Icons.error_outline_rounded : Icons.check_circle_rounded,
          color: Colors.white,
          size: 18,
        ),
        const SizedBox(width: 10),
        Expanded(
            child: Text(msg,
                style:
                const TextStyle(fontFamily: 'Outfit', fontSize: 13))),
      ]),
      backgroundColor: error ? Colors.red.shade700 : Colors.green.shade600,
      behavior: SnackBarBehavior.floating,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
      margin: const EdgeInsets.all(16),
    ));
  }

  // ─────────────────────────────────────────────────────────
  //  BUILD
  // ─────────────────────────────────────────────────────────
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: _surface,
      floatingActionButton: AnimatedBuilder(
        animation: _tabController,
        builder: (_, __) {
          final show = _tabController.index == 1 ||
              (_tabController.animation?.value != null &&
                  (_tabController.animation!.value - 1).abs() < 0.5);
          return AnimatedScale(
            scale: show ? 1.0 : 0.0,
            duration: const Duration(milliseconds: 200),
            child: FloatingActionButton.extended(
              onPressed: _showCreateUserDialog,
              backgroundColor: _blue,
              elevation: 4,
              icon: const Icon(Icons.person_add_rounded,
                  color: Colors.white, size: 20),
              label: const Text('Create Account',
                  style: TextStyle(
                    color: Colors.white,
                    fontFamily: 'Outfit',
                    fontWeight: FontWeight.w700,
                    fontSize: 13,
                  )),
            ),
          );
        },
      ),
      body: Column(
        children: [
          _buildHeader(),
          _buildTabBar(),
          Expanded(
            child: TabBarView(
              controller: _tabController,
              children: [
                _buildOverviewTab(),
                _buildUsersTab(),
                _buildOrdersTab(),
              ],
            ),
          ),
        ],
      ),
    );
  }

  // ─────────────────────────────────────────────────────────
  //  CREATE USER DIALOG
  // ─────────────────────────────────────────────────────────
  void _showCreateUserDialog() {
    final nameCtrl     = TextEditingController();
    final emailCtrl    = TextEditingController();
    final passCtrl     = TextEditingController();
    final mobileCtrl   = TextEditingController();
    String selectedRole = 'user';
    bool obscurePass    = true;
    bool isLoading      = false;

    final formKey = GlobalKey<FormState>();

    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (ctx) {
        return StatefulBuilder(builder: (ctx, setModalState) {
          return Container(
            padding: EdgeInsets.only(
              bottom: MediaQuery.of(ctx).viewInsets.bottom,
            ),
            decoration: const BoxDecoration(
              color: Colors.white,
              borderRadius:
              BorderRadius.vertical(top: Radius.circular(28)),
            ),
            child: SingleChildScrollView(
              padding: const EdgeInsets.fromLTRB(24, 0, 24, 32),
              child: Form(
                key: formKey,
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Center(
                      child: Container(
                        margin: const EdgeInsets.only(top: 12, bottom: 20),
                        width: 40, height: 4,
                        decoration: BoxDecoration(
                          color: Colors.grey.shade200,
                          borderRadius: BorderRadius.circular(2),
                        ),
                      ),
                    ),
                    Row(
                      children: [
                        Container(
                          width: 38, height: 38,
                          decoration: BoxDecoration(
                            color: _blue.withOpacity(0.1),
                            borderRadius: BorderRadius.circular(10),
                          ),
                          child: const Icon(Icons.person_add_rounded,
                              color: _blue, size: 20),
                        ),
                        const SizedBox(width: 12),
                        const Text('Create Account',
                            style: TextStyle(
                              fontSize: 18,
                              fontWeight: FontWeight.w800,
                              color: _navy,
                              fontFamily: 'Outfit',
                            )),
                      ],
                    ),
                    const SizedBox(height: 20),
                    const Text('Account Role',
                        style: TextStyle(
                            fontSize: 12,
                            fontWeight: FontWeight.w600,
                            color: Color(0xFF6B7280),
                            fontFamily: 'Outfit')),
                    const SizedBox(height: 8),
                    Row(
                      children: ['user', 'vendor', 'admin'].map((r) {
                        final active = selectedRole == r;
                        Color rc;
                        switch (r) {
                          case 'admin':  rc = Colors.purple.shade600; break;
                          case 'vendor': rc = Colors.green.shade600;  break;
                          default:       rc = _blue;
                        }
                        return Expanded(
                          child: GestureDetector(
                            onTap: () =>
                                setModalState(() => selectedRole = r),
                            child: AnimatedContainer(
                              duration: const Duration(milliseconds: 150),
                              margin: EdgeInsets.only(right: r != 'admin' ? 8 : 0),
                              padding: const EdgeInsets.symmetric(vertical: 10),
                              decoration: BoxDecoration(
                                color: active ? rc : rc.withOpacity(0.08),
                                borderRadius: BorderRadius.circular(12),
                                border: Border.all(
                                  color: active ? rc : rc.withOpacity(0.2),
                                ),
                              ),
                              child: Column(
                                children: [
                                  Icon(
                                    r == 'admin'
                                        ? Icons.admin_panel_settings_rounded
                                        : r == 'vendor'
                                        ? Icons.store_rounded
                                        : Icons.person_rounded,
                                    color: active ? Colors.white : rc,
                                    size: 18,
                                  ),
                                  const SizedBox(height: 3),
                                  Text(
                                    r[0].toUpperCase() + r.substring(1),
                                    style: TextStyle(
                                      fontSize: 11,
                                      fontWeight: FontWeight.w700,
                                      color: active ? Colors.white : rc,
                                      fontFamily: 'Outfit',
                                    ),
                                  ),
                                ],
                              ),
                            ),
                          ),
                        );
                      }).toList(),
                    ),
                    const SizedBox(height: 16),
                    _dialogField(
                      controller: nameCtrl,
                      label: 'Full Name',
                      hint: 'e.g. Rahul Sharma',
                      icon: Icons.person_outline_rounded,
                      validator: (v) => (v == null || v.isEmpty)
                          ? 'Name is required'
                          : null,
                    ),
                    const SizedBox(height: 12),
                    _dialogField(
                      controller: emailCtrl,
                      label: 'Email Address',
                      hint: 'user@example.com',
                      icon: Icons.alternate_email_rounded,
                      keyboard: TextInputType.emailAddress,
                      validator: (v) {
                        if (v == null || v.isEmpty) return 'Email is required';
                        if (!v.contains('@')) return 'Enter valid email';
                        return null;
                      },
                    ),
                    const SizedBox(height: 12),
                    _dialogField(
                      controller: mobileCtrl,
                      label: 'Mobile Number',
                      hint: '9876543210',
                      icon: Icons.phone_outlined,
                      keyboard: TextInputType.phone,
                      prefixText: '+91  ',
                    ),
                    const SizedBox(height: 12),
                    _dialogField(
                      controller: passCtrl,
                      label: 'Password',
                      hint: 'Min. 6 characters',
                      icon: Icons.lock_outline_rounded,
                      obscure: obscurePass,
                      suffixIcon: GestureDetector(
                        onTap: () => setModalState(() => obscurePass = !obscurePass),
                        child: Icon(
                          obscurePass
                              ? Icons.visibility_off_outlined
                              : Icons.visibility_outlined,
                          color: Colors.grey.shade400,
                          size: 18,
                        ),
                      ),
                      validator: (v) {
                        if (v == null || v.isEmpty) return 'Password is required';
                        if (v.length < 6) return 'Minimum 6 characters';
                        return null;
                      },
                    ),
                    const SizedBox(height: 24),
                    SizedBox(
                      width: double.infinity,
                      height: 50,
                      child: isLoading
                          ? Container(
                        decoration: BoxDecoration(
                          gradient: const LinearGradient(colors: [_navy, _blue]),
                          borderRadius: BorderRadius.circular(14),
                        ),
                        child: const Center(
                          child: SizedBox(
                            width: 20, height: 20,
                            child: CircularProgressIndicator(
                                color: Colors.white, strokeWidth: 2.5),
                          ),
                        ),
                      )
                          : ElevatedButton(
                        onPressed: () async {
                          if (!formKey.currentState!.validate()) return;
                          setModalState(() => isLoading = true);
                          try {
                            final cred = await FirebaseAuth.instance
                                .createUserWithEmailAndPassword(
                              email: emailCtrl.text.trim(),
                              password: passCtrl.text.trim(),
                            );
                            await FirebaseFirestore.instance
                                .collection('users')
                                .doc(cred.user!.uid)
                                .set({
                              'name':     nameCtrl.text.trim(),
                              'email':    emailCtrl.text.trim(),
                              'mobile':   mobileCtrl.text.trim(),
                              'address':  '',
                              'role':     selectedRole,
                              'isBanned': false,
                              'createdAt': FieldValue.serverTimestamp(),
                            });
                            if (mounted) {
                              Navigator.pop(ctx);
                              _snack(
                                  '${selectedRole[0].toUpperCase()}${selectedRole.substring(1)} account created!');
                            }
                          } on FirebaseAuthException catch (e) {
                            setModalState(() => isLoading = false);
                            _snack(_friendlyError(e.code), error: true);
                          } catch (e) {
                            setModalState(() => isLoading = false);
                            _snack('Error: $e', error: true);
                          }
                        },
                        style: ElevatedButton.styleFrom(
                          padding: EdgeInsets.zero,
                          shape: RoundedRectangleBorder(
                              borderRadius: BorderRadius.circular(14)),
                          elevation: 0,
                        ),
                        child: Ink(
                          decoration: BoxDecoration(
                            gradient: const LinearGradient(
                              colors: [_navy, _blue],
                              begin: Alignment.centerLeft,
                              end: Alignment.centerRight,
                            ),
                            borderRadius: BorderRadius.circular(14),
                          ),
                          child: const Center(
                            child: Row(
                              mainAxisSize: MainAxisSize.min,
                              children: [
                                Icon(Icons.check_rounded, color: Colors.white, size: 18),
                                SizedBox(width: 8),
                                Text('Create Account',
                                    style: TextStyle(
                                      color: Colors.white,
                                      fontSize: 14,
                                      fontWeight: FontWeight.w700,
                                      fontFamily: 'Outfit',
                                    )),
                              ],
                            ),
                          ),
                        ),
                      ),
                    ),
                  ],
                ),
              ),
            ),
          );
        });
      },
    );
  }

  Widget _dialogField({
    required TextEditingController controller,
    required String label,
    required String hint,
    required IconData icon,
    TextInputType keyboard = TextInputType.text,
    String? prefixText,
    bool obscure = false,
    Widget? suffixIcon,
    String? Function(String?)? validator,
  }) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(label,
            style: const TextStyle(
                fontSize: 12,
                fontWeight: FontWeight.w600,
                color: Color(0xFF6B7280),
                fontFamily: 'Outfit')),
        const SizedBox(height: 6),
        TextFormField(
          controller: controller,
          keyboardType: keyboard,
          obscureText: obscure,
          validator: validator,
          style: const TextStyle(fontSize: 14, color: _navy, fontFamily: 'Outfit'),
          decoration: InputDecoration(
            hintText: hint,
            hintStyle: TextStyle(
                color: Colors.grey.shade400, fontFamily: 'Outfit', fontSize: 13),
            prefixText: prefixText,
            prefixIcon: Icon(icon, color: Colors.grey.shade400, size: 18),
            suffixIcon: suffixIcon,
            filled: true,
            fillColor: Colors.grey.shade50,
            contentPadding:
            const EdgeInsets.symmetric(horizontal: 16, vertical: 13),
            border: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: BorderSide(color: Colors.grey.shade200),
            ),
            enabledBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: BorderSide(color: Colors.grey.shade200),
            ),
            focusedBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: const BorderSide(color: _blue, width: 1.5),
            ),
            errorBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: BorderSide(color: Colors.red.shade400),
            ),
            focusedErrorBorder: OutlineInputBorder(
              borderRadius: BorderRadius.circular(12),
              borderSide: BorderSide(color: Colors.red.shade400, width: 1.5),
            ),
          ),
        ),
      ],
    );
  }

  String _friendlyError(String code) {
    switch (code) {
      case 'email-already-in-use':
        return 'An account already exists with this email.';
      case 'invalid-email':
        return 'Please enter a valid email address.';
      case 'weak-password':
        return 'Password must be at least 6 characters.';
      case 'too-many-requests':
        return 'Too many attempts. Please try again later.';
      default:
        return 'Something went wrong. Please try again.';
    }
  }

  // ─────────────────────────────────────────────────────────
  //  TAB BAR
  // ─────────────────────────────────────────────────────────
  Widget _buildTabBar() {
    return Container(
      color: Colors.white,
      child: TabBar(
        controller: _tabController,
        labelColor: _blue,
        unselectedLabelColor: Colors.grey.shade400,
        labelStyle: const TextStyle(
            fontWeight: FontWeight.w700, fontFamily: 'Outfit', fontSize: 13),
        unselectedLabelStyle: const TextStyle(
            fontWeight: FontWeight.w500, fontFamily: 'Outfit', fontSize: 13),
        indicatorColor: _blue,
        indicatorWeight: 2.5,
        indicatorSize: TabBarIndicatorSize.label,
        tabs: _tabs.map((t) => Tab(text: t)).toList(),
      ),
    );
  }

  //  TAB 1 — OVERVIEW

        Row(
          mainAxisAlignment: MainAxisAlignment.spaceBetween,
          children: [
            Text(label,
                style: const TextStyle(

                    fontWeight: FontWeight.w700,

        const SizedBox(height: 6),

          borderRadius: BorderRadius.circular(4),

            value: pct,
            minHeight: 6,

  Future<void> _handleUserAction(
      String action, String uid, bool isBanned, String role) async {
    try {
      HapticFeedback.lightImpact();
      final ref = FirebaseFirestore.instance.collection('users').doc(uid);
      switch (action) {
        case 'ban':
          await ref.update({'isBanned': !isBanned});
          _snack(isBanned ? 'User unbanned' : 'User banned');
          break;
        case 'make_vendor':
          await ref.update({'role': role == 'vendor' ? 'user' : 'vendor'});
          _snack(role == 'vendor' ? 'Demoted to User' : 'Promoted to Vendor');
          break;
        case 'make_admin':
          await ref.update({'role': 'admin'});
          _snack('User promoted to Admin');
          break;
        case 'delete':
          final confirmed = await _confirmDelete(context, uid);
          if (!confirmed) return;
          await ref.delete();
          _snack('Account deleted');
          break;
      }
    } catch (e) {
      _snack('Error: $e', error: true);
    }
  }

  String _serviceEmoji(String service) {
    final s = service.toLowerCase();
    if (s.contains('clean'))  return '🧹';
    if (s.contains('plumb') || s.contains('leak') || s.contains('pipe')) return '🔧';
    if (s.contains('ac') || s.contains('gas'))   return '❄️';
    if (s.contains('paint') || s.contains('wallpaper')) return '🖌️';
    if (s.contains('furni') || s.contains('upholster')) return '🪑';
    if (s.contains('pest') || s.contains('termite')) return '🦟';
    return '🏠';
  }

                duration: const Duration(milliseconds: 300),

  // ────────────────────────────────────────────────
  // NAVIGATION LOGIC
  // ────────────────────────────────────────────────
  void _handleNext() {
    if (_currentStep == 0) {
      if (!_validateDetails()) return;
      setState(() => _currentStep = 1);
    } else if (_currentStep == 1) {
      setState(() => _currentStep = 2);
    } else if (_currentStep == 2) {
      _confirmBooking();
    }
  }

  bool _validateDetails() {
    if (selectedDate == null || selectedTime == null) {
      _snack('Please select a date and time');
      return false;
    }
    if (phoneController.text.isEmpty ||
        !RegExp(r'^\d{10}$').hasMatch(phoneController.text)) {
      _snack('Enter a valid 10-digit mobile number');
      return false;
    }
    if (addressController.text.trim().isEmpty) {
      _snack('Please enter the service address');
      return false;
    }
    if (descriptionController.text.trim().isEmpty) {
      _snack('Please describe your requirement');
      return false;
    }
    return true;
  }

  Future<void> _confirmBooking() async {
    setState(() => _isLoading = true);
    try {
      final user = FirebaseAuth.instance.currentUser;
      if (user == null) {
        Navigator.pushReplacementNamed(context, '/login');
        return;
      }
      _orderId = await OrderManager.addOrder(
        userId: user.uid,
        service: widget.serviceName,
        price: widget.price,
        date: '${selectedDate!.day}/${selectedDate!.month}/${selectedDate!.year}',
        time: selectedTime!.format(context),
        address: addressController.text,
        description: descriptionController.text,
        phone: phoneController.text,
      );
      setState(() {
        _isLoading = false;
        _currentStep = 3;
      });
      _checkAnimController.forward();
    } catch (e) {
      setState(() => _isLoading = false);
      _snack('Error: $e');
    }
  }

  static const Color _navy    = Color(0xFF0D2137);
  static const Color _blue    = Color(0xFF1565C0);

  final _textController = TextEditingController();
  final _scrollController = ScrollController();
  final _focusNode = FocusNode();
  String _myUid = '';
  bool _isSending = false;
  bool _hasText = false;

  // typing indicator animation
  late AnimationController _typingDotCtrl;

  @override
  void initState() {
    super.initState();
    _myUid = FirebaseAuth.instance.currentUser?.uid ?? '';
    _typingDotCtrl = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1200),
    )..repeat();
    _textController.addListener(() {
      final has = _textController.text.trim().isNotEmpty;
      if (has != _hasText) setState(() => _hasText = has);
    });
    // mark messages read when screen opens
    _markRead();
  }

    _textController.dispose();
    _scrollController.dispose();
    _focusNode.dispose();
    _typingDotCtrl.dispose();
    super.dispose();
  }

  CollectionReference get _chatRef => FirebaseFirestore.instance
      .collection('orders')
      .doc(widget.orderId)
      .collection('messages');

  Future<void> _markRead() async {
    // optionally update unread count in order doc
    try {
      await FirebaseFirestore.instance
          .collection('orders')
          .doc(widget.orderId)
          .update({'unread_$_myUid': 0});
    } catch (_) {}
  }

  Future<void> _sendMessage() async {
    final text = _textController.text.trim();
    if (text.isEmpty || _isSending) return;
    HapticFeedback.lightImpact();
    setState(() => _isSending = true);
    _textController.clear();
    try {
      await _chatRef.add({
        'text':      text,
        'senderId':  _myUid,
        'senderName': ProfileManager.userName,
        'timestamp': FieldValue.serverTimestamp(),
        'type':      'text',
      });
      _scrollToBottom();
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(SnackBar(
          content: Text('Failed to send: $e',
              style: const TextStyle(fontFamily: 'Outfit')),
          backgroundColor: Colors.red.shade700,
          behavior: SnackBarBehavior.floating,
          shape:
          RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
          margin: const EdgeInsets.all(16),
        ));
      }
    } finally {
      if (mounted) setState(() => _isSending = false);
    }
  }

  void _scrollToBottom({bool animated = true}) {
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (_scrollController.hasClients) {
        if (animated) {
          _scrollController.animateTo(
            _scrollController.position.maxScrollExtent + 80,
            duration: const Duration(milliseconds: 300),
            curve: Curves.easeOut,
          );
        } else {
          _scrollController.jumpTo(
              _scrollController.position.maxScrollExtent + 80);
        }
      }
    });
  }

      builder: (ctx, snap) {
        if (snap.connectionState == ConnectionState.waiting) {
          return const Center(
              child: CircularProgressIndicator(
                  color: _blue, strokeWidth: 2.5));
        }

        final docs = snap.data?.docs ?? [];

        if (docs.isEmpty) {
          return _buildEmptyChat();
        }

        // auto-scroll on new message
        WidgetsBinding.instance.addPostFrameCallback((_) => _scrollToBottom());

        return ListView.builder(
          controller: _scrollController,
          padding: const EdgeInsets.fromLTRB(16, 12, 16, 8),
          itemCount: docs.length,
          itemBuilder: (_, i) {
            final doc  = docs[i];
            final data = doc.data() as Map<String, dynamic>;
            final isMe = (data['senderId'] ?? '') == _myUid;

            // show date divider if this is first msg or date changed
            bool showDate = false;
            if (i == 0) {
              showDate = true;
            } else {
              final prevData =
              docs[i - 1].data() as Map<String, dynamic>;
              final prevTs =
              (prevData['timestamp'] as Timestamp?)?.toDate();
              final curTs =
              (data['timestamp'] as Timestamp?)?.toDate();
              if (prevTs != null && curTs != null) {
                showDate = curTs.day != prevTs.day ||
                    curTs.month != prevTs.month;
              }
            }

            return Column(
              children: [
                if (showDate) _buildDateDivider(
                    (data['timestamp'] as Timestamp?)?.toDate()),
                FadeInUp(
                  duration: const Duration(milliseconds: 200),
                  child: _buildBubble(data, isMe),
                ),
              ],
            );
          },
        );
      },
    );
  }

  Widget _buildEmptyChat() {
    return Center(
      child: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        children: [
          Container(
            width: 80, height: 80,
            decoration: BoxDecoration(
              color: _blue.withOpacity(0.08),
              shape: BoxShape.circle,
            ),
            child: Icon(Icons.chat_bubble_outline_rounded,

                fontFamily: 'Outfit',
              )),

              style: TextStyle(
                fontSize: 13,
                color: Colors.grey.shade400,
                fontFamily: 'Outfit',
              )),
        ],
      ),
    );
  }

  Widget _buildDateDivider(DateTime? date) {
    String label = 'Today';
    if (date != null) {
      final now = DateTime.now();
      final diff = DateTime(now.year, now.month, now.day)
          .difference(DateTime(date.year, date.month, date.day))
          .inDays;
      if (diff == 1) {
        label = 'Yesterday';
      } else if (diff > 1) {
        label =
        '${date.day} ${_monthName(date.month)} ${date.year}';
      }
    }
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 16),
      child: Row(
        children: [
          Expanded(
              child: Divider(
                  color: Colors.grey.shade200, thickness: 1)),
          Padding(
            padding: const EdgeInsets.symmetric(horizontal: 12),
            child: Container(
              padding: const EdgeInsets.symmetric(
                  horizontal: 12, vertical: 4),
              decoration: BoxDecoration(
                color: Colors.white,
                borderRadius: BorderRadius.circular(20),
                border:
                Border.all(color: Colors.grey.shade200),
              ),
              child: Text(label,
                  style: TextStyle(
                    fontSize: 11,
                    fontWeight: FontWeight.w600,
                    color: Colors.grey.shade400,
                    fontFamily: 'Outfit',
                  )),
            ),
          ),
          Expanded(
              child: Divider(
                  color: Colors.grey.shade200, thickness: 1)),
        ],
      ),
    );
  }

  String _monthName(int m) => const [
    '', 'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
    'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'
  ][m];

  // ─────────────────────────────────────────────────────────

      alignment: isMe ? Alignment.centerRight : Alignment.centerLeft,
      child: Container(
        margin: EdgeInsets.only(
          top: 2, bottom: 2,
          left:  isMe ? 60 : 0,

  // ── Submit ─────────────────────────────────────────────────
  Future<void> _submitForm() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() => _isSubmitting = true);

    try {
      await FirebaseFirestore.instance
          .collection('contact_messages')
          .add({
        'name':      _nameCtrl.text.trim(),
        'email':     _emailCtrl.text.trim(),
        'message':   _messageCtrl.text.trim(),
        'timestamp': FieldValue.serverTimestamp(),
        'status':    'unread',
      });

      if (!mounted) return;
      _showSuccessDialog();
    } on FirebaseException catch (e) {
      if (!mounted) return;
      _showErrorDialog(_firestoreErrorMessage(e.code));
    } catch (e) {
      if (!mounted) return;
      _showErrorDialog(
          'Unable to send your message right now.\nPlease try again later.');
    } finally {
      if (mounted) setState(() => _isSubmitting = false);
    }
  }

  // ── Friendly error messages ────────────────────────────────
  String _firestoreErrorMessage(String code) {
    switch (code) {
      case 'permission-denied':
        return 'Permission denied. Please make sure you are logged in and try again.';
      case 'unavailable':
        return 'Service is temporarily unavailable. Please check your internet connection and try again.';
      case 'deadline-exceeded':
        return 'The request timed out. Please check your connection and try again.';
      case 'unauthenticated':
        return 'You need to be logged in to send a message.';
      case 'resource-exhausted':
        return 'Too many requests. Please wait a moment and try again.';
      default:
        return 'Unable to send your message right now.\nPlease try again later.';
    }
  }

  Future<void> _checkBannedStatus() async {
    final user = FirebaseAuth.instance.currentUser;
    if (user != null) {
      final userDoc = await FirebaseFirestore.instance
          .collection('users')
          .doc(user.uid)
          .get();
      if (userDoc.exists && userDoc['isBanned'] == true) {
        await FirebaseAuth.instance.signOut();
        if (mounted) {
          Navigator.pushReplacementNamed(context, '/login');
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('Your account is banned.')),
          );
        }
      }
    }
  }

  Future<void> _logout() async {
    await FirebaseAuth.instance.signOut();
    if (mounted) {
      Navigator.pushReplacementNamed(context, '/login');
    }

class _DotGridPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = Colors.white.withOpacity(0.06)
      ..style = PaintingStyle.fill;

    const spacing = 20.0;
    const radius = 1.5;

    for (double x = spacing; x < size.width; x += spacing) {
      for (double y = spacing; y < size.height; y += spacing) {
        canvas.drawCircle(Offset(x, y), radius, paint);
      }
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

  void initState() {
    super.initState();
    _emailFocus.addListener(() => setState(() => _emailFocused = _emailFocus.hasFocus));
    _passFocus.addListener(()  => setState(() => _passFocused  = _passFocus.hasFocus));

    _emailController.dispose();

  Future<void> _login() async {
    if (!_formKey.currentState!.validate()) return;
    HapticFeedback.lightImpact();
    setState(() => _isLoading = true);
    try {
      final cred = await FirebaseAuth.instance.signInWithEmailAndPassword(
        email:    _emailController.text.trim(),
        password: _passwordController.text.trim(),
      );
      final userDoc = await FirebaseFirestore.instance
          .collection('users').doc(cred.user!.uid).get();

      if (!userDoc.exists) {
        await _createUserDoc(cred.user!.uid);
        await _redirect('user');
        return;
      }
      final data = userDoc.data()!;
      if (data['isBanned'] == true) {
        await FirebaseAuth.instance.signOut();
        _showError('Your account has been suspended.\nContact support.');
        return;
      }
      await userDoc.reference.update({'lastLogin': FieldValue.serverTimestamp()});
      _loadProfile(data);
      await _redirect(data['role'] ?? 'user');
    } on FirebaseAuthException catch (e) {
      _showError(_friendlyError(e.code));
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Future<void> _signUp() async {
    if (!_formKey.currentState!.validate()) return;
    HapticFeedback.lightImpact();
    setState(() => _isLoading = true);
    try {
      final cred = await FirebaseAuth.instance.createUserWithEmailAndPassword(
        email:    _emailController.text.trim(),
        password: _passwordController.text.trim(),
      );
      await _createUserDoc(cred.user!.uid);
      await _redirect('user');
    } on FirebaseAuthException catch (e) {
      _showError(_friendlyError(e.code));
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Future<void> _createUserDoc(String uid) async {
    await FirebaseFirestore.instance.collection('users').doc(uid).set({
      'name': 'User', 'email': _emailController.text.trim(),
      'mobile': '', 'address': '', 'role': 'user',

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:uuid/uuid.dart';

class OrderManager {
  static final _firestore = FirebaseFirestore.instance;
  static final _ordersRef = _firestore.collection('orders');

  static Future<void> init() async {}

  static Future<String> addOrder({
    required String userId,
    required String service,
    required String price,
    required String date,
    required String time,
    required String address,
    required String description,
    required String phone,
  }) async {
    final orderId = const Uuid().v4();
    await _ordersRef.doc(orderId).set({
      'id': orderId,
      'userId': userId,
      'service': service,
      'price': price,
      'date': date,
      'time': time,
      'address': address,
      'description': description,
      'phone': phone,
      'status': 'pending',
      'createdAt': FieldValue.serverTimestamp(),
    });
    return orderId;
  }

  static Future<void> updateStatus(String orderId, String status) async {
    await _ordersRef.doc(orderId).update({'status': status});
  }

  static Stream<QuerySnapshot> vendorOrdersStream() {
    return _ordersRef.snapshots();
  }

  static Stream<QuerySnapshot> userOrdersStream(String userId) {
    return _ordersRef.where('userId', isEqualTo: userId).snapshots();
  }
}

  // ── orders list ───────────────────────────────────────────
  Widget _buildOrdersList(String userId) {
    return StreamBuilder<QuerySnapshot>(
      stream: OrderManager.userOrdersStream(userId),
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Center(
              child: CircularProgressIndicator(color: _blue, strokeWidth: 2.5));
        }

        if (snapshot.hasError) {
          return _errorState(snapshot.error.toString());
        }

        if (!snapshot.hasData || snapshot.data!.docs.isEmpty) {
          return _emptyState();
        }

        // filter docs
        var docs = snapshot.data!.docs;
        if (_filter != 'All') {
          docs = docs
              .where((d) =>
          ((d.data() as Map)['status'] ?? 'pending').toString().toLowerCase() ==
              _filter.toLowerCase())
              .toList();
        }

        if (docs.isEmpty) return _emptyState(filtered: true);

        return ListView.builder(
          padding: const EdgeInsets.fromLTRB(16, 8, 16, 24),
          itemCount: docs.length,
          itemBuilder: (context, index) {
            final doc  = docs[index];
            final data = doc.data() as Map<String, dynamic>;
            final status  = (data['status'] ?? 'pending').toString().toLowerCase();
            final service = data['service'] ?? 'Service';

            return FadeInUp(
              duration: Duration(milliseconds: 300 + index * 60),
              child: _buildOrderCard(
                doc: doc,
                data: data,
                status: status,
                service: service,
                index: index,
              ),
            );
          },
        );
      },
    );
  }

// lib/profile_manager.dart
import 'dart:io';
import 'package:shared_preferences/shared_preferences.dart';

class ProfileManager {
  static File? profilePhoto;
  static String userName = 'User';
  static String email = '';
  static String mobile = '';
  static String address = '';
  static String role = 'user'; // user | vendor | admin

  static Future loadProfile() async {
    final prefs = await SharedPreferences.getInstance();
    userName = prefs.getString('userName') ?? 'User';
    email = prefs.getString('email') ?? '';
    mobile = prefs.getString('mobile') ?? '';
    address = prefs.getString('address') ?? '';
    role = prefs.getString('role') ?? 'user';
  }

  static Future saveProfile() async {
    final prefs = await SharedPreferences.getInstance();
    await Future.wait([
      prefs.setString('userName', userName),
      prefs.setString('email', email),
      prefs.setString('mobile', mobile),
      prefs.setString('address', address),
      prefs.setString('role', role),
    ]);
  }

  static void updateField(String field, String value) {
    switch (field) {
      case 'name': userName = value; break;
      case 'email': email = value; break;
      case 'mobile': mobile = value; break;
      case 'address': address = value; break;
      case 'role': role = value; break;
    }
    saveProfile();
  }
}

// ─────────────────────────────────────────────────────────────
//  DOT GRID PATTERN PAINTER
// ─────────────────────────────────────────────────────────────
class _DotGridPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    const spacing = 22.0;
    final paint = Paint()
      ..color = Colors.white
      ..style = PaintingStyle.fill;

    for (double x = 0; x < size.width; x += spacing) {
      for (double y = 0; y < size.height; y += spacing) {
        canvas.drawCircle(Offset(x, y), 1.5, paint);
      }
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class SplashScreen extends StatefulWidget {
  const SplashScreen({super.key});

  @override
  State<SplashScreen> createState() => _SplashScreenState();
}

class _SplashScreenState extends State<SplashScreen>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _animation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      duration: const Duration(seconds: 2),
      vsync: this,
    );
    _animation = Tween<double>(begin: 0.0, end: 1.0).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeInOut),
    );
    _controller.forward();

    Future.delayed(const Duration(seconds: 5), () {
      if (mounted) {
        Navigator.pushReplacementNamed(context, '/login');
      }
    });
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFF1E40AF),
      body: Center(
        child: FadeTransition(
          opacity: _animation,
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              SizedBox(
                width: 80,
                height: 80,

                  'assets/logo.png',

  final _addressController = TextEditingController();

  bool _isSaving  = false;

  final _nameFocus    = FocusNode();
  final _emailFocus   = FocusNode();
  final _mobileFocus  = FocusNode();
  final _addressFocus = FocusNode();

  static const Color _navy    = Color(0xFF0D2137);
  static const Color _blue    = Color(0xFF1565C0);
  static const Color _surface = Color(0xFFF4F7FB);

  @override
  void initState() {
    super.initState();
    _loadProfile();
    for (final fn in [_nameFocus, _emailFocus, _mobileFocus, _addressFocus]) {
      fn.addListener(() => setState(() {}));
    }
  }

  void _loadProfile() {
    _nameController.text    = ProfileManager.userName;
    _emailController.text   = ProfileManager.email;
    _mobileController.text  = ProfileManager.mobile;
    _addressController.text = ProfileManager.address;
  }

  @override
  void dispose() {
    for (final c in [
      _nameController,
      _emailController,
      _mobileController,
      _addressController
    ]) {
      c.dispose();
    }
    for (final fn in [
      _nameFocus,
      _emailFocus,
      _mobileFocus,
      _addressFocus
    ]) {
      fn.dispose();
    }
    super.dispose();
  }

  // ── SAVE — writes to both ProfileManager AND Firestore ───
  Future<void> _saveProfile() async {
    setState(() => _isSaving = true);
    HapticFeedback.lightImpact();

    final name    = _nameController.text.trim();
    final email   = _emailController.text.trim();
    final mobile  = _mobileController.text.trim();
    final address = _addressController.text.trim();

    // 1. Update local ProfileManager
    ProfileManager.updateField('name',    name);
    ProfileManager.updateField('email',   email);
    ProfileManager.updateField('mobile',  mobile);
    ProfileManager.updateField('address', address);
    await ProfileManager.saveProfile();

    // 2. Update Firestore so the drawer sees the new name immediately
    try {
      final uid = FirebaseAuth.instance.currentUser?.uid;
      if (uid != null) {
        await FirebaseFirestore.instance
            .collection('users')
            .doc(uid)
            .update({
          'name':    name,
          'email':   email,
          'mobile':  mobile,
          'address': address,
        });
      }
    } catch (e) {
      // Firestore update failed — local save still succeeded
      debugPrint('Firestore update error: $e');
    }

    if (!mounted) return;
    setState(() {
      _isSaving  = false;
      _isEditing = false;
    });

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: const Row(children: [
          Icon(Icons.check_circle_rounded, color: Colors.white, size: 18),
          SizedBox(width: 10),
          Text('Profile updated!',
              style: TextStyle(fontFamily: 'Outfit', fontSize: 14)),
        ]),
        backgroundColor: Colors.green.shade600,
        behavior: SnackBarBehavior.floating,
        shape:
        RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
        margin: const EdgeInsets.all(16),
        duration: const Duration(seconds: 2),
      ),
    );
  }

  // ─────────────────────────────────────────────────────────
  //  ORDERS LIST
  // ─────────────────────────────────────────────────────────
  Widget _buildOrdersList(String selectedCategory) {
    final currentUid = FirebaseAuth.instance.currentUser?.uid;

    return StreamBuilder<QuerySnapshot>(
      stream: FirebaseFirestore.instance
          .collection('orders')
          .orderBy('createdAt', descending: true)
          .snapshots(),
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Center(
              child: CircularProgressIndicator(
                  color: _blue, strokeWidth: 2.5));
        }

        if (snapshot.hasError) {
          return _errorState(snapshot.error.toString());
        }

        if (!snapshot.hasData || snapshot.data!.docs.isEmpty) {
          return _emptyState(selectedCategory);
        }

        var docs = snapshot.data!.docs;

        // category filter
        if (selectedCategory != 'All') {
          docs = docs.where((doc) {
            final data = doc.data() as Map<String, dynamic>?;
            if (data == null) return false;
            final serviceRaw = (data['service'] as String?)?.trim() ?? '';
            if (serviceRaw == selectedCategory) return true;
            if (serviceToCategory.containsKey(serviceRaw)) {
              return serviceToCategory[serviceRaw] == selectedCategory;
            }
            return serviceRaw.toLowerCase()
                .contains(selectedCategory.toLowerCase());
          }).toList();
        }

        if (docs.isEmpty) return _emptyState(selectedCategory);

        final total    = docs.length;
        final pending  = docs.where((d) =>
        ((d.data() as Map)['status'] ?? '') == 'pending').length;
        final accepted = docs.where((d) =>
        ((d.data() as Map)['status'] ?? '') == 'accepted').length;

        return Column(
          children: [
            _buildStatsStrip(total, pending, accepted),
            Expanded(
              child: ListView.builder(
                padding: const EdgeInsets.fromLTRB(16, 8, 16, 24),
                itemCount: docs.length,
                itemBuilder: (context, index) {
                  final doc  = docs[index];
                  final data = doc.data() as Map<String, dynamic>;
                  final status  =
                  (data['status'] ?? 'pending').toString().toLowerCase();
                  final service = data['service'] ?? 'Unknown Service';
                  final isMyAcceptedOrder =
                      status == 'accepted' && data['vendorId'] == currentUid;

                  return FadeInUp(
                    duration:
                    Duration(milliseconds: 280 + index * 55),
                    child: _buildOrderCard(
                      doc: doc,
                      data: data,
                      status: status,
                      service: service,
                      isMyAcceptedOrder: isMyAcceptedOrder,
                    ),
                  );
                },
              ),
            ),
          ],
        );
      },
    );
  }

  Widget _actionBtn({
    required IconData icon,
    required Color color,
    required Color bg,
    required VoidCallback onTap,
  }) {
    return GestureDetector(
      onTap: onTap,
      child: Container(
        width: 36, height: 36,
        decoration: BoxDecoration(
          color: bg,
          shape: BoxShape.circle,
        ),
        child: Icon(icon, color: color, size: 18),
      ),
    );
  }

  Widget _detailRow(IconData icon, String label, String value) {
    return Row(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Container(
          width: 30, height: 30,
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(8),
          ),
          child: Icon(icon, size: 15, color: _blue),
        ),
        const SizedBox(width: 10),
        Expanded(
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text(label,
                  style: TextStyle(
                    fontSize: 11,
                    color: Colors.grey.shade500,
                    fontFamily: 'Outfit',
                  )),
              const SizedBox(height: 1),
              Text(value,
                  style: const TextStyle(
                    fontSize: 13,
                    fontWeight: FontWeight.w600,
                    color: _navy,
                    fontFamily: 'Outfit',
                  )),
            ],
          ),
        ),
      ],
    );
  }
